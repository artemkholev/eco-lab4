# Отчет по лабораторной работе №42
## Реализация вытесняющего планировщика задач с алгоритмом LJF (Longest Job First)

![alt text](image.png)

**Longest Job First (LJF) с вытеснением (Preemptive)** - алгоритм планирования, при котором процессор всегда выполняет задачу с максимальной **оставшейся** длительностью. При этом задачи могут прерываться таймером для переключения на более приоритетные:

1. Из списка доступных задач выбирается задача с максимальной **оставшейся** длительностью (`remaining_duration`)
2. Задача выполняется до следующего прерывания таймера (вытесняющий алгоритм)
3. При прерывании:
   - Сохраняется контекст текущей задачи
   - Уменьшается `remaining_duration` на величину интервала таймера
   - Выбирается новая задача с максимальной оставшейся длительностью
   - Восстанавливается контекст выбранной задачи
4. Процесс повторяется до завершения всех задач

## Исходный код реализации компонента

### Структура задачи (CEcoTask1Lab.h)

```c
typedef struct CEcoTask1Lab_C761620F {
    IEcoTask1VTbl* m_pVTblITask;
    uint32_t m_cRef;

    /* Стек */
    volatile byte_t* m_sp;

    /* Данные задачи */
    void (*pfunc) (uint64_t);  // Указатель на функцию задачи
    uint64_t duration;         // Начальная длительность выполнения

    /* Для вытесняющей многозадачности */
    uint64_t remaining_duration;  // Оставшаяся длительность
    uint8_t is_finished;           // Флаг завершения задачи
} CEcoTask1Lab_C761620F;
```

**Ключевые поля для вытеснения:**
- `m_sp` - указатель на стек задачи (для сохранения контекста)
- `remaining_duration` - отслеживание прогресса выполнения
- `is_finished` - пометка завершенных задач

### Ключевой код планировщика (CEcoTaskScheduler1Lab.c)

#### Создание задачи

```c
#define MAX_STATIC_TASK_COUNT 10

int16_t ECOCALLMETHOD CEcoTaskScheduler1Lab_C761620F_NewTask(
    IEcoTaskScheduler1Ptr_t me,
    voidptr_t address,
    voidptr_t data,
    uint32_t stackSize,
    IEcoTask1** ppITask
) {
    int32_t indx = 0;
    int32_t reg = 30;
    uint64_t* pxTopOfStack = 0;

    /* Проверяем указатель пула статических задач */
    for (indx = 0; indx < MAX_STATIC_TASK_COUNT; indx++) {
        if (g_xCEcoTask1List_C761620F[indx].pfunc == 0) {
            g_xCEcoTask1List_C761620F[indx].pfunc = address;
            g_xCEcoTask1List_C761620F[indx].duration = (uint64_t)data;

            /* Инициализация для вытесняющей многозадачности */
            g_xCEcoTask1List_C761620F[indx].remaining_duration = (uint64_t)data;
            g_xCEcoTask1List_C761620F[indx].is_finished = 0;

            g_xCEcoTask1List_C761620F[indx].m_cRef = 1;
            g_xCEcoTask1List_C761620F[indx].m_sp = (byte_t*)&g_xCEcoStackTask1List_C761620F[indx*4096];

            /* Инициализация стека задачи */
            pxTopOfStack = g_xCEcoTask1List_C761620F[indx].m_sp;
            while (reg > 0) {
                pxTopOfStack--;
                reg--;
            }
            *pxTopOfStack = (uint64_t)g_xCEcoTask1List_C761620F[indx].pfunc;
            *ppITask = (IEcoTask1*)&g_xCEcoTask1List_C761620F[indx];
            return 0;
        }
    }
    return -1;
}
```

#### Функция переключения контекста (TaskSwitchContext)

**Ключевая функция вытесняющего LJF** - вызывается при каждом прерывании таймера:

```c
void CEcoTaskScheduler1Lab_C761620F_TaskSwitchContext(void) {
    size_t i = 0;
    uint64_t maxRemainingDuration = 0;
    int32_t maxIndex = -1;
    uint64_t currentTaskIndx = g_indx;

    /* Уменьшаем remaining_duration для текущей задачи */
    if (g_xCEcoTask1List_C761620F[currentTaskIndx].pfunc != 0 &&
        !g_xCEcoTask1List_C761620F[currentTaskIndx].is_finished) {
        if (g_xCEcoTask1List_C761620F[currentTaskIndx].remaining_duration > g_timerInterval) {
            g_xCEcoTask1List_C761620F[currentTaskIndx].remaining_duration -= g_timerInterval;
        } else {
            /* Задача завершена */
            g_xCEcoTask1List_C761620F[currentTaskIndx].remaining_duration = 0;
            g_xCEcoTask1List_C761620F[currentTaskIndx].is_finished = 1;
        }
    }

    /* LJF: выбираем задачу с максимальной оставшейся длительностью */
    for (i = 0; i < MAX_STATIC_TASK_COUNT; ++i) {
        if (g_xCEcoTask1List_C761620F[i].pfunc != 0 &&
            !g_xCEcoTask1List_C761620F[i].is_finished &&
            g_xCEcoTask1List_C761620F[i].remaining_duration > maxRemainingDuration) {
            maxRemainingDuration = g_xCEcoTask1List_C761620F[i].remaining_duration;
            maxIndex = i;
        }
    }

    /* Переключаемся на выбранную задачу */
    if (maxIndex >= 0) {
        g_indx = maxIndex;
        g_pxCurrentTCB_C761620F = (uint64_t*)&g_xCEcoTask1List_C761620F[g_indx];
    }

    /* Сброс таймера для следующего прерывания */
    if (g_xCEcoTaskScheduler1Lab_C761620F.m_pIArmTimer != 0) {
        g_xCEcoTaskScheduler1Lab_C761620F.m_pIArmTimer->pVTbl->Reset(
            g_xCEcoTaskScheduler1Lab_C761620F.m_pIArmTimer);
    }
}
```

#### Запуск планировщика

```c
int16_t ECOCALLMETHOD CEcoTaskScheduler1Lab_C761620F_Start(
    IEcoTaskScheduler1Ptr_t me
) {
    CEcoTaskScheduler1Lab_C761620F* pCMe = (CEcoTaskScheduler1Lab_C761620F*)me;
    size_t i = 0;
    uint64_t maxDuration = 0;
    int32_t maxIndex = -1;

    /* Поиск задачи с максимальной начальной длительностью */
    for (i = 0; i < MAX_STATIC_TASK_COUNT; ++i) {
        if (pCMe->m_pTaskList[i].pfunc != 0 &&
            pCMe->m_pTaskList[i].duration > maxDuration) {
            maxIndex = i;
            maxDuration = pCMe->m_pTaskList[i].duration;
        }
    }

    /* Установка первой задачи (с максимальной длительностью) */
    if (maxIndex >= 0) {
        g_indx = maxIndex;
        g_pxCurrentTCB_C761620F = (uint64_t*)&pCMe->m_pTaskList[maxIndex];
    }

    /* ЗАПУСК ТАЙМЕРА для вытесняющей многозадачности */
    pCMe->m_pIArmTimer->pVTbl->Start(pCMe->m_pIArmTimer);

    /* Запуск первой задачи - дальше управление переходит к таймеру */
    if (pCMe->m_pTaskList[g_indx].pfunc != 0) {
        pCMe->m_pTaskList[g_indx].pfunc(pCMe->m_pTaskList[g_indx].duration);
    }

    /* Бесконечный цикл ожидания - задачи выполняются через прерывания */
    while (1) {
        __asm__ volatile ("WFI");
    }

    return 0;
}
```

## Выводы

### Характеристики вытесняющего алгоритма LJF

**Преимущества:**
- **Настоящая многозадачность** - задачи выполняются параллельно через вытеснение
- **Динамический приоритет** - учитывается оставшееся время выполнения
- **Отзывчивость системы** - короткие задачи могут завершиться раньше
- **Справедливое распределение** процессорного времени между задачами
- **Гибкость** - возможность прерывания и возобновления задач

**Недостатки:**
- Более **сложная реализация** по сравнению с невытесняющим вариантом
- **Накладные расходы** на сохранение/восстановление контекста
- Короткие задачи всё равно могут ждать долго в начале
- Требует **аппаратной поддержки** прерываний
